private: System::Void button1_Click(System::Object^ sender, System::EventArgs^ e) {
	std::list<double> list = { 3.5, -4.2, 15.0, -1.0, 7.1, -8.3 };

	listBox1->Items->Clear();
	for (double x : list) {
		listBox1->Items->Add(x.ToString());
	}

	bool found = false;
	double firstPositive = 0;
	for (double x : list) {
		if (x > 0) {
			firstPositive = x;
			found = true;
			break;
		}
	}

	if (found) {
		for (auto it = list.begin(); it != list.end(); ++it) {
			if (*it < 0) {
				list.insert(std::next(it), firstPositive);
				++it; 
			}
		}
	}

	listBox2->Items->Clear();
	for (double x : list) {
		listBox2->Items->Add(x.ToString());
	}
}
